  Q102.0  color=CueColorData(palette=('amber', 'white'), saturation='높음', palette_id=None, source='director')
  Q103.0  color=CueColorData(palette=('magenta', 'red'), saturation='최대(단색 볼드)', palette_id=None, source='director')

    ClearAll
    Fixture 1 + 2 + 3 + 4 ; At Preset 2.4
    Fixture 1 + 2 + 3 + 4 ; Attribute 'Dimmer' At 90
    Fixture 1 + 2 + 3 + 4 ; Attribute 'ColorRGB_R' At 100 ; Attribute 'ColorRGB_G' At 55 ; Attribute 'ColorRGB_B' At 5
    Store Sequence 210 Cue 102 'Verse' CueFade 0.75
    ClearAll
    Fixture 1 + 2 + 3 + 4 ; At Preset 2.4
    Fixture 1 + 2 + 3 + 4 ; Attribute 'Dimmer' At 100
    Fixture 1 + 2 + 3 + 4 ; Attribute 'ColorRGB_R' At 100 ; Attribute 'ColorRGB_G' At 0 ; Attribute 'ColorRGB_B' At 70
    Store Sequence 210 Cue 103 'Chorus' CueFade 0.25
    ClearAll
    Store Timecode 9
    Set Timecode 9 Property 'Name' 'Probe Seq Timecode'

    Set Cue 101 Sequence 210 Property 'TrigType' 'Time'
    Set Cue 101 Sequence 210 Property 'TrigTime' 0
    Set Cue 102 Sequence 210 Property 'TrigType' 'Time'
    Set Cue 102 Sequence 210 Property 'TrigTime' 20
